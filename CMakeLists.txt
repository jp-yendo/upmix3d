cmake_minimum_required(VERSION 3.24)
project(upmix3d VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(VLC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/external/vlc-3.0.24" CACHE PATH "VLC portable distribution containing sdk and libvlccore.dll")
option(UPMIX3D_BUILD_PLUGIN "Build the Windows VLC output module" ${WIN32})

add_library(upmix3d_core STATIC src/core.cpp src/transport.cpp src/limiter.cpp src/profiles.cpp src/boundary.cpp src/speaker_test.cpp src/messages.cpp)
target_include_directories(upmix3d_core PUBLIC include)
if(MSVC)
    target_compile_options(upmix3d_core PRIVATE /W4 /permissive- /utf-8 /EHsc)
else()
    target_compile_options(upmix3d_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(upmix3d_tests tests/core_tests.cpp)
    target_link_libraries(upmix3d_tests PRIVATE upmix3d_core)
    if(MSVC)
        target_compile_options(upmix3d_tests PRIVATE /W4 /utf-8 /EHsc)
    endif()
    add_test(NAME upmix3d_core COMMAND upmix3d_tests)
    add_executable(upmix3d_benchmark tests/benchmark.cpp)
    target_link_libraries(upmix3d_benchmark PRIVATE upmix3d_core)
endif()

if(UPMIX3D_BUILD_PLUGIN)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The VLC module requires 64-bit Windows")
    endif()
    if(NOT EXISTS "${VLC_ROOT}/sdk/include/vlc/plugins/vlc_aout.h")
        message(FATAL_ERROR "Set VLC_ROOT to a VLC 3.0.x x64 portable distribution containing sdk/")
    endif()
    # The SDK contains an MSVC import library. No hand-written ABI structures or exported entrypoint aliases.
    add_library(upmix3d MODULE src/vlc_module.cpp src/windows_audio.cpp)
    target_include_directories(upmix3d PRIVATE include src "${VLC_ROOT}/sdk/include/vlc" "${VLC_ROOT}/sdk/include/vlc/plugins")
    target_compile_definitions(upmix3d PRIVATE __PLUGIN__ MODULE_STRING="upmix3d" MODULE_SYMBOL=3_0_0f
        WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(upmix3d PRIVATE upmix3d_core "${VLC_ROOT}/sdk/lib/libvlccore.lib" ole32 uuid avrt propsys)
    set_target_properties(upmix3d PROPERTIES PREFIX "lib" OUTPUT_NAME "upmix3d_plugin")
    if(MSVC)
        target_compile_options(upmix3d PRIVATE /W4 /permissive- /utf-8 /EHsc)
    endif()
    if(BUILD_TESTING)
        add_executable(upmix3d_integration tests/windows_integration.cpp)
        target_include_directories(upmix3d_integration PRIVATE "${VLC_ROOT}/sdk/include")
        target_link_libraries(upmix3d_integration PRIVATE dbghelp)
        if(MSVC)
            target_compile_options(upmix3d_integration PRIVATE /W4 /utf-8 /EHsc)
        endif()
    endif()
    install(TARGETS upmix3d LIBRARY DESTINATION plugins/audio_output RUNTIME DESTINATION plugins/audio_output)
endif()
